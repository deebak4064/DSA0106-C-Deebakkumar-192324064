CLASS StudentRecord
    PRIVATE:
        INTEGER   rollNo
        STRING    studentName
        ARRAY     marks[3] OF REAL
        REAL      percentage
        CHARACTER grade

        METHOD CalculateResult()                 // private helper
            percentage ← (marks[0] + marks[1] + marks[2]) / 3.0
            IF percentage >= 85.0 THEN
                grade ← 'A'
            ELSE IF percentage >= 70.0 THEN
                grade ← 'B'
            ELSE IF percentage >= 50.0 THEN
                grade ← 'C'
            ELSE
                grade ← 'F'
            ENDIF
        ENDMETHOD

    PUBLIC:
        CONSTRUCTOR StudentRecord()              // default constructor
            rollNo      ← 0
            studentName ← "-"
            marks[0] ← 0.0
            marks[1] ← 0.0
            marks[2] ← 0.0
            percentage  ← 0.0
            grade       ← '-'
        ENDCONSTRUCTOR

        METHOD ReadData(INTEGER r, STRING name, REAL m1, REAL m2, REAL m3)
            rollNo      ← r
            studentName ← name
            marks[0] ← m1
            marks[1] ← m2
            marks[2] ← m3
            CalculateResult()
        ENDMETHOD

        METHOD GetPercentage() RETURNS REAL
            RETURN percentage
        ENDMETHOD

        METHOD Display()
            PRINT rollNo, studentName, marks[0], marks[1], marks[2], percentage, grade
        ENDMETHOD
ENDCLASS

BEGIN Main
    CONSTANT INTEGER N ← 3
    ARRAY roster[N] OF StudentRecord             // needs the default constructor
    INTEGER pass, j, i
    BOOLEAN swapped
    StudentRecord temp

    roster[0].ReadData(101, "Aarav",  65, 70, 72)
    roster[1].ReadData(102, "Bhavna", 88, 92, 85)
    roster[2].ReadData(103, "Chirag", 75, 80, 70)

    // Bubble sort, descending order of percentage
    FOR pass ← 1 TO N - 1 STEP 1
        swapped ← FALSE
        FOR j ← 0 TO N - pass - 1 STEP 1
            IF roster[j].GetPercentage() < roster[j + 1].GetPercentage() THEN
                temp          ← roster[j]        // whole-object swap
                roster[j]     ← roster[j + 1]
                roster[j + 1] ← temp
                swapped ← TRUE
                PRINT "Pass ", pass, ": swapped positions ", j, " and ", j + 1
            ENDIF
        ENDFOR
        IF swapped = FALSE THEN
            PRINT "Pass ", pass, ": no swaps, sorting complete"
            BREAK
        ENDIF
    ENDFOR

    PRINT "Rank Roll Name M1 M2 M3 Percent Grade"
    FOR i ← 0 TO N - 1 STEP 1
        PRINT i + 1
        roster[i].Display()
    ENDFOR
END
